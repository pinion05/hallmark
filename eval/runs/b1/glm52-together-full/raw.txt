(function() {
  // Play buttons toggle aria-pressed; no audio file is loaded — these are sample slots.
  document.querySelectorAll('.play-btn').forEach(function(btn) {
    btn.addEventListener('click', function() {
      var pressed = btn.getAttribute('aria-pressed') === 'true';
      btn.setAttribute('aria-pressed', pressed ? 'false' : 'true');
      btn.querySelector('span').textContent = pressed ? 'Play' : 'Stop';
    });
  });

  // Form: minimal validation, no celebratory toast.
  var form = document.querySelector('.studio__form');
  if (form) {
    form.addEventListener('submit', function(e) {
      e.preventDefault();
      var email = form.querySelector('#email');
      var emailErr = form.querySelector('#email-error');
      var valid = true;
      emailErr.textContent = '';
      email.classList.remove('input--error');
      if (!email.value || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email.value)) {
        emailErr.textContent = 'Please enter a valid email address.';
        email.classList.add('input--error');
        email.focus({ preventScroll: false });
        valid = false;
      }
      if (valid) {
        var btn = form.querySelector('.btn--primary');
        btn.textContent = 'Sent — we\'ll reply within a day';
        btn.setAttribute('aria-disabled', 'true');
        btn.disabled = true;
      }
    });
  }
})();
</script>
