<script>
(function () {
  "use strict";

  const VOICES = [
    { name: "Kick",  desc: "60hz sine · fast pitch drop",  type: "kick" },
    { name: "Snare", desc: "noise burst · 200hz tone",     type: "snare" },
    { name: "Hat",   desc: "highpass noise · 7khz",        type: "hat" },
    { name: "Open",  desc: "noise · longer decay",        type: "open" },
    { name: "Clap",  desc: "three noise bursts",          type: "clap" },
    { name: "Tom",   desc: "100hz sine · mid decay",      type: "tom" },
    { name: "Rim",   desc: "short click · 1.2khz",        type: "rim" },
    { name: "Cymb",  desc: "noise · bright · long tail",  type: "cymbal" }
  ];

  const STEPS = 16;
  const ROWS = VOICES.length;

  let pattern = VOICES.map(() => Array(STEPS).fill(false));
  let muted = VOICES.map(() => false);
  let bpm = 96;
  let isPlaying = false;
  let currentStep = 0;
  let nextNoteTime = 0;
  let timerID = null;
  let lookahead = 25.0;
  let scheduleAheadTime = 0.1;
  let lastDrawnStep = -1;

  const knobs = {
    swing: { value: 0, min: 0, max: 50, label: "swing" },
    vol:   { value: 75, min: 0, max: 100, label: "volume" },
    rev:   { value: 18, min: 0, max: 100, label: "reverb" }
  };

  let audioCtx = null;
  let masterGain = null;
  let reverbNode = null;
  let reverbWet = null;
  let reverbDry = null;

  function ensureAudio() {
    if (audioCtx) return;
    const Ctx = window.AudioContext || window.webkitAudioContext;
    audioCtx = new Ctx();
    masterGain = audioCtx.createGain();
    masterGain.gain.value = knobs.vol.value / 100;

    // simple reverb via feedback delay
    reverbNode = audioCtx.createDelay(1.0);
    reverbNode.delayTime.value = 0.18;
    const fb = audioCtx.createGain();
    fb.gain.value = 0.35;
    reverbWet = audioCtx.createGain();
    reverbDry = audioCtx.createGain();
    reverbWet.gain.value = knobs.rev.value / 100;
    reverbDry.gain.value = 1;

    reverbNode.connect(fb);
    fb.connect(reverbNode);
    reverbNode.connect(reverbWet);

    masterGain.connect(reverbDry);
    masterGain.connect(reverbNode);
    reverbDry.connect(audioCtx.destination);
    reverbWet.connect(audioCtx.destination);
  }

  function noiseBuffer(dur) {
    const len = Math.floor(audioCtx.sampleRate * dur);
    const buf = audioCtx.createBuffer(1, len, audioCtx.sampleRate);
    const data = buf.getChannelData(0);
    for (let i = 0; i < len; i++) data[i] = Math.random() * 2 - 1;
    return buf;
  }

  function playVoice(voiceIdx, time) {
    if (muted[voiceIdx]) return;
    const v = VOICES[voiceIdx];
    if (!audioCtx) return;

    const out = masterGain;

    if (v.type === "kick") {
      const osc = audioCtx.createOscillator();
      const g = audioCtx.createGain();
      osc.frequency.setValueAtTime(120, time);
      osc.frequency.exponentialRampToValueAtTime(40, time + 0.12);
      g.gain.setValueAtTime(0.9, time);
      g.gain.exponentialRampToValueAtTime(0.001, time + 0.3);
      osc.connect(g); g.connect(out);
      osc.start(time); osc.stop(time + 0.32);
    } else if (v.type === "snare") {
      const noise = audioCtx.createBufferSource();
      noise.buffer = noiseBuffer(0.2);
      const hp = audioCtx.createBiquadFilter();
      hp.type = "highpass"; hp.frequency.value = 1500;
      const ng = audioCtx.createGain();
      ng.gain.setValueAtTime(0.5, time);
      ng.gain.exponentialRampToValueAtTime(0.001, time + 0.18);
      noise.connect(hp); hp.connect(ng); ng.connect(out);
      noise.start(time); noise.stop(time + 0.2);

      const tone = audioCtx.createOscillator();
      const tg = audioCtx.createGain();
      tone.frequency.value = 200;
      tg.gain.setValueAtTime(0.3, time);
      tg.gain.exponentialRampToValueAtTime(0.001, time + 0.1);
      tone.connect(tg); tg.connect(out);
      tone.start(time); tone.stop(time + 0.12);
    } else if (v.type === "hat") {
      const noise = audioCtx.createBufferSource();
      noise.buffer = noiseBuffer(0.05);
      const hp = audioCtx.createBiquadFilter();
      hp.type = "highpass"; hp.frequency.value = 7000;
      const g = audioCtx.createGain();
      g.gain.setValueAtTime(0.35, time);
      g.gain.exponentialRampToValueAtTime(0.001, time + 0.04);
      noise.connect(hp); hp.connect(g); g.connect(out);
      noise.start(time); noise.stop(time + 0.06);
    } else if (v.type === "open") {
      const noise = audioCtx.createBufferSource();
      noise.buffer = noiseBuffer(0.4);
      const hp = audioCtx.createBiquadFilter();
      hp.type = "highpass"; hp.frequency.value = 6000;
      const g = audioCtx.createGain();
      g.gain.setValueAtTime(0.3, time);
      g.gain.exponentialRampToValueAtTime(0.001, time + 0.3);
      noise.connect(hp); hp.connect(g); g.connect(out);
      noise.start(time); noise.stop(time + 0.4);
    } else if (v.type === "clap") {
      const offsets = [0, 0.012, 0.024, 0.04];
      offsets.forEach((off, i) => {
        const noise = audioCtx.createBufferSource();
        noise.buffer = noiseBuffer(0.08);
        const bp = audioCtx.createBiquadFilter();
        bp.type = "bandpass"; bp.frequency.value = 1200; bp.Q.value = 1.2;
        const g = audioCtx.createGain();
        const t = time + off;
        const peak = i === offsets.length - 1 ? 0.45 : 0.25;
        g.gain.setValueAtTime(peak, t);
        g.gain.exponentialRampToValueAtTime(0.001, t + 0.08);
        noise.connect(bp); bp.connect(g); g.connect(out);
        noise.start(t); noise.stop(t + 0.1);
      });
    } else if (v.type === "tom") {
      const osc = audioCtx.createOscillator();
      const g = audioCtx.createGain();
      osc.frequency.setValueAtTime(180, time);
      osc.frequency.exponentialRampToValueAtTime(100, time + 0.2);
      g.gain.setValueAtTime(0.6, time);
      g.gain.exponentialRampToValueAtTime(0.001, time + 0.25);
      osc.connect(g); g.connect(out);
      osc.start(time); osc.stop(time + 0.27);
    } else if (v.type === "rim") {
      const osc = audioCtx.createOscillator();
      const g = audioCtx.createGain();
      osc.type = "square";
      osc.frequency.value = 1200;
      g.gain.setValueAtTime(0.3, time);
      g.gain.exponentialRampToValueAtTime(0.001, time + 0.02);
      osc.connect(g); g.connect(out);
      osc.start(time); osc.stop(time + 0.03);
    } else if (v.type === "cymbal") {
      const noise = audioCtx.createBufferSource();
      noise.buffer = noiseBuffer(0.8);
      const hp = audioCtx.createBiquadFilter();
      hp.type = "highpass"; hp.frequency.value = 5000;
      const g = audioCtx.createGain();
      g.gain.setValueAtTime(0.2, time);
      g.gain.exponentialRampToValueAtTime(0.001, time + 0.7);
      noise.connect(hp); hp.connect(g); g.connect(out);
      noise.start(time); noise.stop(time + 0.8);
    }
  }

  function nextNote() {
    const secondsPerBeat = 60.0 / bpm;
    const swingAmt = knobs.swing.value / 100;
    const base = secondsPerBeat / 4; // 16th notes
    // swing: delay odd 16ths
    const isOff = currentStep % 2 === 1;
    nextNoteTime += base * (1 + (isOff ? swingAmt : 0));
    currentStep = (currentStep + 1) % STEPS;
  }

  function scheduleNote(beatNumber, time) {
    for (let r = 0; r < ROWS; r++) {
      if (pattern[r][beatNumber]) playVoice(r, time);
    }
  }

  function scheduler() {
    while (nextNoteTime < audioCtx.currentTime + scheduleAheadTime) {
      scheduleNote(currentStep, nextNoteTime);
      nextNote();
    }
    timerID = window.setTimeout(scheduler, lookahead);
  }

  function draw() {
    const stepToDraw = currentStep;
    if (lastDrawnStep === stepToDraw) {
      requestAnimationFrame(draw);
      return;
    }
    lastDrawnStep = stepToDraw;

    // update step readout
    stepCurrentEl.textContent = String(stepToDraw + 1).padStart(2, "0");

    // clear all playing states
    document.querySelectorAll(".cell--playing").forEach(el => el.classList.remove("cell--playing"));
    // mark current step column
    document.querySelectorAll('.cell[data-step="' + stepToDraw + '"]').forEach(el => el.classList.add("cell--playing"));

    // voice meters flash
    for (let r = 0; r < ROWS; r++) {
      if (pattern[r][stepToDraw] && !muted[r]) {
        const meter = document.querySelector('.voice[data-voice="' + r + '"] .voice__meter');
        if (meter) {
          meter.querySelectorAll(".meter-bar").forEach(b => {
            b.classList.add("meter-bar--active");
            b.style.opacity = "1";
          });
          window.setTimeout(() => {
            meter.querySelectorAll(".meter-bar").forEach((b, i) => {
              b.classList.remove("meter-bar--active");
              b.style.opacity = String(1 - (i + 1) * 0.18);
            });
          }, 90);
        }
      }
    }

    requestAnimationFrame(draw);
  }

  // ─── DOM build ───
  const stepRuler = document.getElementById("step-ruler");
  for (let i = 0; i < STEPS; i++) {
    const mark = document.createElement("span");
    mark.className = "seq__step-mark" + (i % 4 === 0 ? " seq__step-mark--downbeat" : "");
    mark.textContent = String(i + 1).padStart(2, "0");
    stepRuler.appendChild(mark);
  }

  const seqRows = document.getElementById("seq-rows");
  VOICES.forEach((v, r) => {
    const row = document.createElement("div");
    row.className = "seq__row";
    row.dataset.voice = r;

    const label = document.createElement("div");
    label.className = "seq__row-label";
    const name = document.createElement("span");
    name.className = "seq__row-name";
    name.textContent = v.name;
    const num = document.createElement("span");
    num.className = "seq__row-num";
    num.textContent = String(r + 1).padStart(2, "0");
    label.appendChild(name);
    label.appendChild(num);

    const cells = document.createElement("div");
    cells.className = "seq__cells";
    for (let s = 0; s < STEPS; s++) {
      const cell = document.createElement("button");
      cell.type = "button";
      cell.className = "cell" + (s % 4 === 0 ? " cell--downbeat" : "");
      cell.dataset.voice = r;
      cell.dataset.step = s;
      cell.setAttribute("aria-label", v.name + " step " + (s + 1));
      cell.setAttribute("aria-pressed", "false");
      const numEl = document.createElement("span");
      numEl.className = "cell__num";
      numEl.textContent = String(s + 1).padStart(2, "0");
      numEl.setAttribute("aria-hidden", "true");
      cell.appendChild(numEl);
      cell.addEventListener("click", () => {
        pattern[r][s] = !pattern[r][s];
        cell.classList.toggle("cell--on", pattern[r][s]);
        cell.setAttribute("aria-pressed", pattern[r][s] ? "true" : "false");
        if (audioCtx && pattern[r][s]) playVoice(r, audioCtx.currentTime);
      });
      cells.appendChild(cell);
    }

    row.appendChild(label);
    row.appendChild(cells);
    seqRows.appendChild(row);
  });

  // Voices list
  const voiceList = document.getElementById("voice-list");
  VOICES.forEach((v, r) => {
    const li = document.createElement("li");
    li.className = "voice";
    li.dataset.voice = r;

    const num = document.createElement("span");
    num.className = "voice__num";
    num.textContent = String(r + 1).padStart(2, "0");

    const main = document.createElement("div");
    main.className = "voice__main";
    const name = document.createElement("span");
    name.className = "voice__name";
    name.textContent = v.name;
    const desc = document.createElement("span");
    desc.className = "voice__desc";
    desc.textContent = v.desc;
    main.appendChild(name);
    main.appendChild(desc);

    const meter = document.createElement("div");
    meter.className = "voice__meter";
    meter.setAttribute("aria-hidden", "true");
    for (let i = 0; i < 5; i++) {
      const bar = document.createElement("span");
      bar.className = "meter-bar";
      bar.style.opacity = String(1 - i * 0.18);
      meter.appendChild(bar);
    }

    const toggle = document.createElement("div");
    toggle.className = "voice__toggle";
    const muteBtn = document.createElement("button");
    muteBtn.type = "button";
    muteBtn.className = "mute-btn";
    muteBtn.textContent = "mute";
    muteBtn.setAttribute("aria-label", "Mute " + v.name);
    muteBtn.setAttribute("aria-pressed", "false");
    muteBtn.addEventListener("click", () => {
      muted[r] = !muted[r];
      muteBtn.classList.toggle("is-muted", muted[r]);
      muteBtn.textContent = muted[r] ? "muted" : "mute";
      muteBtn.setAttribute("aria-pressed", muted[r] ? "true" : "false");
    });
    toggle.appendChild(muteBtn);

    li.appendChild(num);
    li.appendChild(main);
    li.appendChild(meter);
    li.appendChild(toggle);
    voiceList.appendChild(li);
  });

  // Knobs
  const knobRow = document.getElementById("knob-row");
  Object.keys(knobs).forEach(key => {
    const k = knobs[key];
    const knob = document.createElement("div");
    knob.className = "knob";

    const dial = document.createElement("div");
    dial.className = "knob__dial";
    dial.tabIndex = 0;
    dial.setAttribute("role", "slider");
    dial.setAttribute("aria-label", k.label);
    dial.setAttribute("aria-valuemin", String(k.min));
    dial.setAttribute("aria-valuemax", String(k.max));

<script>
(function () {
  "use strict";

  const VOICES = [
    { name: "Kick",  desc: "60hz sine · fast pitch drop",  type: "kick" },
    { name: "Snare", desc: "noise burst · 200hz tone",     type: "snare" },
    { name: "Hat",   desc: "highpass noise · 7khz",        type: "hat" },
    { name: "Open",  desc: "noise · longer decay",         type: "open" },
    { name: "Clap",  desc: "three noise bursts",          type: "clap" },
    { name: "Tom",   desc: "100hz sine · mid decay",       type: "tom" },
    { name: "Rim",   desc: "short click · 1.2khz",        type: "rim" },
    { name: "Cymb",  desc: "noise · bright · long tail",   type: "cymbal" }
  ];

  const STEPS = 16;
  const ROWS = VOICES.length;

  let pattern = VOICES.map(() => Array(STEPS).fill(false));
  let muted = VOICES.map(() => false);
  let bpm = 96;
  let isPlaying = false;
  let currentStep = 0;
  let nextNoteTime = 0;
  let timerID = null;
  const lookahead = 25.0;
  const scheduleAheadTime = 0.1;
  let lastDrawnStep = -1;

  const knobs = {
    swing: { value: 0, min: 0, max: 50, label: "swing" },
    vol:   { value: 75, min: 0, max: 100, label: "volume" },
    rev:   { value: 18, min: 0, max: 100, label: "reverb" }
  };

  let audioCtx = null;
  let masterGain = null;
  let reverbNode = null;
  let reverbWet = null;
  let reverbDry = null;

  function ensureAudio() {
    if (audioCtx) return;
    const Ctx = window.AudioContext || window.webkitAudioContext;
    audioCtx = new Ctx();
    masterGain = audioCtx.createGain();
    masterGain.gain.value = knobs.vol.value / 100;

    reverbNode = audioCtx.createDelay(1.0);
    reverbNode.delayTime.value = 0.18;
    const fb = audioCtx.createGain();
    fb.gain.value = 0.35;
    reverbWet = audioCtx.createGain();
    reverbDry = audioCtx.createGain();
    reverbWet.gain.value = knobs.rev.value / 100;
    reverbDry.gain.value = 1;

    reverbNode.connect(fb);
    fb.connect(reverbNode);
    reverbNode.connect(reverbWet);

    masterGain.connect(reverbDry);
    masterGain.connect(reverbNode);
    reverbDry.connect(audioCtx.destination);
    reverbWet.connect(audioCtx.destination);
  }

  function noiseBuffer(dur) {
    const len = Math.floor(audioCtx.sampleRate * dur);
    const buf = audioCtx.createBuffer(1, len, audioCtx.sampleRate);
    const data = buf.getChannelData(0);
    for (let i = 0; i < len; i++) data[i] = Math.random() * 2 - 1;
    return buf;
  }

  function playVoice(voiceIdx, time) {
    if (muted[voiceIdx]) return;
    const v = VOICES[voiceIdx];
    if (!audioCtx) return;

    const out = masterGain;

    if (v.type === "kick") {
      const osc = audioCtx.createOscillator();
      const g = audioCtx.createGain();
      osc.frequency.setValueAtTime(120, time);
      osc.frequency.exponentialRampToValueAtTime(40, time + 0.12);
      g.gain.setValueAtTime(0.9, time);
      g.gain.exponentialRampToValueAtTime(0.001, time + 0.3);
      osc.connect(g); g.connect(out);
      osc.start(time); osc.stop(time + 0.32);
    } else if (v.type === "snare") {
      const noise = audioCtx.createBufferSource();
      noise.buffer = noiseBuffer(0.2);
      const hp = audioCtx.createBiquadFilter();
      hp.type = "highpass"; hp.frequency.value = 1500;
      const ng = audioCtx.createGain();
      ng.gain.setValueAtTime(0.5, time);
      ng.gain.exponentialRampToValueAtTime(0.001, time + 0.18);
      noise.connect(hp); hp.connect(ng); ng.connect(out);
      noise.start(time); noise.stop(time + 0.2);

      const tone = audioCtx.createOscillator();
      const tg = audioCtx.createGain();
      tone.frequency.value = 200;
      tg.gain.setValueAtTime(0.3, time);
      tg.gain.exponentialRampToValueAtTime(0.001, time + 0.1);
      tone.connect(tg); tg.connect(out);
      tone.start(time); tone.stop(time + 0.12);
    } else if (v.type === "hat") {
      const noise = audioCtx.createBufferSource();
      noise.buffer = noiseBuffer(0.05);
      const hp = audioCtx.createBiquadFilter();
      hp.type = "highpass"; hp.frequency.value = 7000;
      const g = audioCtx.createGain();
      g.gain.setValueAtTime(0.35, time);
      g.gain.exponentialRampToValueAtTime(0.001, time + 0.04);
      noise.connect(hp); hp.connect(g); g.connect(out);
      noise.start(time); noise.stop(time + 0.06);
    } else if (v.type === "open") {
      const noise = audioCtx.createBufferSource();
      noise.buffer = noiseBuffer(0.4);
      const hp = audioCtx.createBiquadFilter();
      hp.type = "highpass"; hp.frequency.value = 6000;
      const g = audioCtx.createGain();
      g.gain.setValueAtTime(0.3, time);
      g.gain.exponentialRampToValueAtTime(0.001, time + 0.3);
      noise.connect(hp); hp.connect(g); g.connect(out);
      noise.start(time); noise.stop(time + 0.4);
    } else if (v.type === "clap") {
      const offsets = [0, 0.012, 0.024, 0.04];
      offsets.forEach((off, i) => {
        const noise = audioCtx.createBufferSource();
        noise.buffer = noiseBuffer(0.08);
        const bp = audioCtx.createBiquadFilter();
        bp.type = "bandpass"; bp.frequency.value = 1200; bp.Q.value = 1.2;
        const g = audioCtx.createGain();
        const t = time + off;
        const peak = i === offsets.length - 1 ? 0.45 : 0.25;
        g.gain.setValueAtTime(peak, t);
        g.gain.exponentialRampToValueAtTime(0.001, t + 0.08);
        noise.connect(bp); bp.connect(g); g.connect(out);
        noise.start(t); noise.stop(t + 0.1);
      });
    } else if (v.type === "tom") {
      const osc = audioCtx.createOscillator();
      const g = audioCtx.createGain();
      osc.frequency.setValueAtTime(180, time);
      osc.frequency.exponentialRampToValueAtTime(100, time + 0.2);
      g.gain.setValueAtTime(0.6, time);
      g.gain.exponentialRampToValueAtTime(0.001, time + 0.25);
      osc.connect(g); g.connect(out);
      osc.start(time); osc.stop(time + 0.27);
    } else if (v.type === "rim") {
      const osc = audioCtx.createOscillator();
      const g = audioCtx.createGain();
      osc.type = "square";
      osc.frequency.value = 1200;
      g.gain.setValueAtTime(0.3, time);
      g.gain.exponentialRampToValueAtTime(0.001, time + 0.02);
      osc.connect(g); g.connect(out);
      osc.start(time); osc.stop(time + 0.03);
    } else if (v.type === "cymbal") {
      const noise = audioCtx.createBufferSource();
      noise.buffer = noiseBuffer(0.8);
      const hp = audioCtx.createBiquadFilter();
      hp.type = "highpass"; hp.frequency.value = 5000;
      const g = audioCtx.createGain();
      g.gain.setValueAtTime(0.2, time);
      g.gain.exponentialRampToValueAtTime(0.001, time + 0.7);
      noise.connect(hp); hp.connect(g); g.connect(out);
      noise.start(time); noise.stop(time + 0.8);
    }
  }

  function nextNote() {
    const secondsPerBeat = 60.0 / bpm;
    const swingAmt = knobs.swing.value / 100;
    const base = secondsPerBeat / 4;
    const isOff = currentStep % 2 === 1;
    nextNoteTime += base * (1 + (isOff ? swingAmt : 0));
    currentStep = (currentStep + 1) % STEPS;
  }

  function scheduleNote(beatNumber, time) {
    for (let r = 0; r < ROWS; r++) {
      if (pattern[r][beatNumber]) playVoice(r, time);
    }
  }

  function scheduler() {
    while (nextNoteTime < audioCtx.currentTime + scheduleAheadTime) {
      scheduleNote(currentStep, nextNoteTime);
      nextNote();
    }
    timerID = window.setTimeout(scheduler, lookahead);
  }

  function draw() {
    const stepToDraw = currentStep;
    if (lastDrawnStep === stepToDraw) {
      requestAnimationFrame(draw);
      return;
    }
    lastDrawnStep = stepToDraw;

    stepCurrentEl.textContent = String(stepToDraw + 1).padStart(2, "0");

    document.querySelectorAll(".cell--playing").forEach(el => el.classList.remove("cell--playing"));
    document.querySelectorAll('.cell[data-step="' + stepToDraw + '"]').forEach(el => el.classList.add("cell--playing"));

    for (let r = 0; r < ROWS; r++) {
      if (pattern[r][stepToDraw] && !muted[r]) {
        const meter = document.querySelector('.voice[data-voice="' + r + '"] .voice__meter');
        if (meter) {
          meter.querySelectorAll(".meter-bar").forEach(b => {
            b.classList.add("meter-bar--active");
            b.style.opacity = "1";
          });
          window.setTimeout(() => {
            meter.querySelectorAll(".meter-bar").forEach((b, i) => {
              b.classList.remove("meter-bar--active");
              b.style.opacity = String(1 - (i + 1) * 0.18);
            });
          }, 90);
        }
      }
    }

    requestAnimationFrame(draw);
  }

  // ─── DOM build ───
  const stepRuler = document.getElementById("step-ruler");
  for (let i = 0; i < STEPS; i++) {
    const mark = document.createElement("span");
    mark.className = "seq__step-mark" + (i % 4 === 0 ? " seq__step-mark--downbeat" : "");
    mark.textContent = String(i + 1).padStart(2, "0");
    stepRuler.appendChild(mark);
  }

  const seqRows = document.getElementById("seq-rows");
  VOICES.forEach((v, r) => {
    const row = document.createElement("div");
    row.className = "seq__row";
    row.dataset.voice = r;

    const label = document.createElement("div");
    label.className = "seq__row-label";
    const name = document.createElement("span");
    name.className = "seq__row-name";
    name.textContent = v.name;
    const num = document.createElement("span");
    num.className = "seq__row-num";
    num.textContent = String(r + 1).padStart(2, "0");
    label.appendChild(name);
    label.appendChild(num);

    const cells = document.createElement("div");
    cells.className = "seq__cells";
    for (let s = 0; s < STEPS; s++) {
      const cell = document.createElement("button");
      cell.type = "button";
      cell.className = "cell" + (s % 4 === 0 ? " cell--downbeat" : "");
      cell.dataset.voice = r;
      cell.dataset.step = s;
      cell.setAttribute("aria-label", v.name + " step " + (s + 1));
      cell.setAttribute("aria-pressed", "false");
      const numEl = document.createElement("span");
      numEl.className = "cell__num";
      numEl.textContent = String(s + 1).padStart(2, "0");
      numEl.setAttribute("aria-hidden", "true");
      cell.appendChild(numEl);
      cell.addEventListener("click", () => {
        pattern[r][s] = !pattern[r][s];
        cell.classList.toggle("cell--on", pattern[r][s]);
        cell.setAttribute("aria-pressed", pattern[r][s] ? "true" : "false");
        if (audioCtx && pattern[r][s]) playVoice(r, audioCtx.currentTime);
      });
      cells.appendChild(cell);
    }

    row.appendChild(label);
    row.appendChild(cells);
    seqRows.appendChild(row);
  });

  const voiceList = document.getElementById("voice-list");
  VOICES.forEach((v, r) => {
    const li = document.createElement("li");
    li.className = "voice";
    li.dataset.voice = r;

    const num = document.createElement("span");
    num.className = "voice__num";
    num.textContent = String(r + 1).padStart(2, "0");

    const main = document.createElement("div");
    main.className = "voice__main";
    const name = document.createElement("span");
    name.className = "voice__name";
    name.textContent = v.name;
    const desc = document.createElement("span");
    desc.className = "voice__desc";
    desc.textContent = v.desc;
    main.appendChild(name);
    main.appendChild(desc);

    const meter = document.createElement("div");
    meter.className = "voice__meter";
    meter.setAttribute("aria-hidden", "true");
    for (let i = 0; i < 5; i++) {
      const bar = document.createElement("span");
      bar.className = "meter-bar";
      bar.style.opacity = String(1 - i * 0.18);
      meter.appendChild(bar);
    }

    const toggle = document.createElement("div");
    toggle.className = "voice__toggle";
    const muteBtn = document.createElement("button");
    muteBtn.type = "button";
    muteBtn.className = "mute-btn";
    muteBtn.textContent = "mute";
    muteBtn.setAttribute("aria-label", "Mute " + v.name);
    muteBtn.setAttribute("aria-pressed", "false");
    muteBtn.addEventListener("click", () => {
      muted[r] = !muted[r];
      muteBtn.classList.toggle("is-muted", muted[r]);
      muteBtn.textContent = muted[r] ? "muted" : "mute";
      muteBtn.setAttribute("aria-pressed", muted[r] ? "true" : "false");
    });
    toggle.appendChild(muteBtn);

    li.appendChild(num);
    li.appendChild(main);
    li.appendChild(meter);
    li.appendChild(toggle);
    voiceList.appendChild(li);
  });

  const knobRow = document.getElementById("knob-row");
  Object.keys(knobs).forEach(key => {
    const k = knobs[key];
    const knob = document.createElement("div");
    knob.className = "knob";

    const dial = document.createElement("div");
    dial.className = "knob__dial";
    dial.tabIndex = 0;
    dial.setAttribute("role", "slider");
    dial.setAttribute("aria-label", k.label);
    dial.setAttribute("aria-valuemin", String(k.min));
    dial.setAttribute("aria-valuemax", String(k.max));
    dial.setAttribute("aria-valuenow", String(k.value));

    const name = document.createElement("span");
    name.className = "knob__name";
    name.textContent = k.label;

    const val = document.createElement("span");
    val.className = "knob__value";
    val.textContent = k.value;

    knob.appendChild(dial);
    knob.appendChild(name);
    knob.appendChild(val);
    knobRow.appendChild(knob);

    function setRot() {
      const pct = (k.value - k.min) / (k.max - k.min);
      const deg = -135 + pct * 270;
      dial.style.setProperty("--rot", deg + "deg");
    }
    setRot();

    function setValue(v) {
      k.value = Math.max(k.min, Math.min(k.max, Math.round(v)));
      val.textContent = k.value;
      dial.setAttribute("aria-valuenow", String(k.value));
      setRot();
      if (key === "vol" && masterGain) masterGain.gain.value = k.value / 100;
      if (key === "rev" && reverbWet) reverbWet.gain.value = k.value / 100;
    }

    let dragging = false;
    let startY = 0;
    let startVal = 0;
    dial.addEventListener("pointerdown", (e) => {
      dragging = true;
      startY = e.clientY;
      startVal = k.value;
      dial.setPointerCapture(e.pointerId);
      e.preventDefault();
    });
    dial.addEventListener("pointermove", (e) => {
      if (!dragging) return;
      const dy = startY - e.clientY;
      const range = k.max - k.min;
      setValue(startVal + (dy / 120) * range);
    });
    dial.addEventListener("pointerup", (e) => {
      dragging = false;
      try { dial.releasePointerCapture(e.pointerId); } catch (_) {}
    });
    dial.addEventListener("pointercancel", () => { dragging = false; });

    dial.addEventListener("keydown", (e) => {
      if (e.key === "ArrowUp" || e.key === "ArrowRight") {
        e.preventDefault();
        setValue(k.value + (e.shiftKey ? 5 : 1));
      } else if (e.key === "ArrowDown" || e.key === "ArrowLeft") {
        e.preventDefault();
        setValue(k.value - (e.shiftKey ? 5 : 1));
      } else if (e.key === "Home") {
        e.preventDefault();
        setValue(k.min);
      } else if (e.key === "End") {
        e.preventDefault();
        setValue(k.max);
      }
    });
  });

  const playBtn = document.getElementById("play-btn");
  const stopBtn = document.getElementById("stop-btn");
  const clearBtn = document.getElementById("clear-btn");
  const bpmUp = document.getElementById("bpm-up");
  const bpmDown = document.getElementById("bpm-down");
  const bpmValue = document.getElementById("bpm-value");
  const stepCurrentEl = document.getElementById("step-current");
  const statusText = document.querySelector(".status-text");
  const masthead = document.querySelector(".masthead");

  function setBpm(v) {
    bpm = Math.max(40, Math.min(220, Math.round(v)));
    bpmValue.textContent = bpm;
  }

  function start() {
    ensureAudio();
    if (audioCtx.state === "suspended") audioCtx.resume();
    isPlaying = true;
    currentStep = 0;
    nextNoteTime = audioCtx.currentTime + 0.05;
    lastDrawnStep = -1;
    scheduler();
    requestAnimationFrame(draw);
    playBtn.classList.add("is-playing");
    playBtn.setAttribute("aria-label", "Pause");
    playBtn.querySelector(".btn__label").textContent = "Pause";
    playBtn.querySelector(".icon").innerHTML = '<rect x="4" y="3" width="3" height="10"/><rect x="9" y="3" width="3" height="10"/>';
    stopBtn.disabled = false;
    statusText.textContent = "playing";
    masthead.classList.add("is-playing");
  }

  function stop() {
    isPlaying = false;
    if (timerID) { window.clearTimeout(timerID); timerID = null; }
    currentStep = 0;
    document.querySelectorAll(".cell--playing").forEach(el => el.classList.remove("cell--playing"));
    stepCurrentEl.textContent = "—";
    playBtn.classList.remove("is-playing");
    playBtn.setAttribute("aria-label", "Play");
    playBtn.querySelector(".btn__label").textContent = "Play";
    playBtn.querySelector(".icon").innerHTML = '<path d="M4 3l9 5-9 5z"/>';
    stopBtn.disabled = true;
    statusText.textContent = "stopped";
    masthead.classList.remove("is-playing");
  }

  playBtn.addEventListener("click", () => { if (isPlaying) stop(); else start(); });
  stopBtn.addEventListener("click", stop);
  clearBtn.addEventListener("click", () => {
    pattern = VOICES.map(() => Array(STEPS).fill(false));
    document.querySelectorAll(".cell").forEach(c => {
      c.classList.remove("cell--on");
      c.setAttribute("aria-pressed", "false");
    });
  });
  bpmUp.addEventListener("click", (e) => setBpm(bpm + (e.shiftKey ? 5 : 1)));
  bpmDown.addEventListener("click", (e) => setBpm(bpm - (e.shiftKey ? 5 : 1)));

  document.addEventListener("keydown", (e) => {
    if (e.code === "Space" && e.target.tagName !== "BUTTON" && e.target.tagName !== "INPUT") {
      e.preventDefault();
      if (isPlaying) stop(); else start();
    }
  });

  // Seed a starter pattern — four-on-the-floor with hats on the offbeats
  const seed = {
    0: [0, 4, 8, 12],
    1: [4, 12],
    2: [0, 2, 4, 6, 8, 10, 12, 14],
    4: [7, 15]
  };
  Object.keys(seed).forEach(r => {
    seed[r].forEach(s => {
      pattern[r][s] = true;
      const cell = document.querySelector('.cell[data-voice="' + r + '"][data-step="' + s + '"]');
      if (cell) {
        cell.classList.add("cell--on");
        cell.setAttribute("aria-pressed", "true");
      }
    });
  });

})();
